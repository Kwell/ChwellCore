cmake_minimum_required(VERSION 3.14)
project(ChwellClusterReference VERSION 1.0.0 LANGUAGES CXX)
if(NOT TARGET Chwell::core)
    find_package(ChwellCore 0.1 CONFIG REQUIRED COMPONENTS core schema consul)
    if(NOT ChwellCore_WITH_MYSQL)
        message(FATAL_ERROR "Cluster reference requires ChwellCore built with MySQL support")
    endif()
endif()
find_package(nlohmann_json 3.11 REQUIRED)
chwell_generate_entity_schema(cluster_player player.schema.json generated/cluster_player.h)
add_executable(cluster_reference main.cpp)
add_dependencies(cluster_reference cluster_player)
target_include_directories(cluster_reference PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(cluster_reference PRIVATE Chwell::schema Chwell::consul nlohmann_json::nlohmann_json)
target_compile_features(cluster_reference PRIVATE cxx_std_17)
target_compile_options(cluster_reference PRIVATE -Wall -Wextra)
