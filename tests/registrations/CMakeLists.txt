cmake_minimum_required(VERSION 3.14)
project(ChwellRegistrationTests LANGUAGES CXX)
find_package(GTest QUIET)
if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG f8d7d77c06936315286eb55f8de22cd23c188571)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
endif()
find_package(Threads REQUIRED)
add_executable(chwell_registration_tests ../test_registrations.cpp ../../src/core/timer_wheel.cpp
    ../../src/core/logger.cpp)
target_compile_features(chwell_registration_tests PRIVATE cxx_std_17)
target_include_directories(chwell_registration_tests PRIVATE ../../include)
target_link_libraries(chwell_registration_tests PRIVATE GTest::gtest_main Threads::Threads)
if(MSVC)
    target_compile_options(chwell_registration_tests PRIVATE /utf-8 /W4)
else()
    target_compile_options(chwell_registration_tests PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME chwell_registration_tests COMMAND chwell_registration_tests)
set_tests_properties(chwell_registration_tests PROPERTIES TIMEOUT 30)
