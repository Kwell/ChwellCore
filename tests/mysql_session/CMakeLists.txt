cmake_minimum_required(VERSION 3.14)
project(MysqlSessionContract LANGUAGES CXX)
find_package(ChwellCore 0.1 CONFIG REQUIRED COMPONENTS core schema)
if(NOT ChwellCore_WITH_MYSQL)
    message(FATAL_ERROR "Real session contract requires MySQL")
endif()
add_executable(mysql_session_contract main.cpp)
target_link_libraries(mysql_session_contract PRIVATE Chwell::core)
target_compile_features(mysql_session_contract PRIVATE cxx_std_17)
add_executable(mysql_migration_contract migration.cpp)
target_link_libraries(mysql_migration_contract PRIVATE Chwell::schema)
target_compile_features(mysql_migration_contract PRIVATE cxx_std_17)
add_executable(mysql_idempotency_contract idempotency.cpp)
target_link_libraries(mysql_idempotency_contract PRIVATE Chwell::core)
target_compile_features(mysql_idempotency_contract PRIVATE cxx_std_17)
