cmake_minimum_required(VERSION 3.14)
project(ChwellRuntimePluginTests LANGUAGES CXX)
find_package(GTest QUIET)
if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG f8d7d77c06936315286eb55f8de22cd23c188571)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
endif()
find_package(Threads REQUIRED)
add_executable(chwell_runtime_plugin_tests ../test_runtime_plugin.cpp ../../src/service/runtime_plugin.cpp)
target_include_directories(chwell_runtime_plugin_tests PRIVATE ../../include)
target_compile_features(chwell_runtime_plugin_tests PRIVATE cxx_std_17)
target_link_libraries(chwell_runtime_plugin_tests PRIVATE GTest::gtest_main Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(chwell_runtime_plugin_tests PRIVATE /utf-8 /W4)
else()
    target_compile_options(chwell_runtime_plugin_tests PRIVATE -Wall -Wextra)
endif()
include(fixtures.cmake)
chwell_runtime_plugin_fixtures(chwell_runtime_plugin_tests)
enable_testing()
add_test(NAME chwell_runtime_plugin_tests COMMAND chwell_runtime_plugin_tests)
set_tests_properties(chwell_runtime_plugin_tests PROPERTIES TIMEOUT 30)
