cmake_minimum_required(VERSION 3.18)
project(ChwellHttpClientContract LANGUAGES C CXX)
enable_testing()
find_package(Threads REQUIRED)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
option(CHWELL_TEST_FETCH_CURL "Fetch pinned curl for isolated Windows tests" OFF)
if(CHWELL_TEST_FETCH_CURL)
    include(FetchContent)
    set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
    set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
    set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
    set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
    set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
    set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
    set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
    set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
    set(CURL_DISABLE_LDAP ON CACHE BOOL "" FORCE)
    set(CURL_DISABLE_LDAPS ON CACHE BOOL "" FORCE)
    FetchContent_Declare(curl
        GIT_REPOSITORY https://github.com/curl/curl.git
        GIT_TAG 57495c64871d18905a0941db9196ef90bafe9a29)
    FetchContent_MakeAvailable(curl)
else()
    find_package(CURL 7.68 REQUIRED)
endif()
add_library(chwell_http_client ../../src/http/http_client.cpp)
target_include_directories(chwell_http_client PUBLIC ../../include)
target_compile_features(chwell_http_client PUBLIC cxx_std_17)
target_link_libraries(chwell_http_client PRIVATE CURL::libcurl)
add_executable(chwell_http_client_contract main.cpp)
target_link_libraries(chwell_http_client_contract PRIVATE chwell_http_client Threads::Threads)
if(MSVC)
    target_compile_options(chwell_http_client PRIVATE /W4 /utf-8)
    target_compile_options(chwell_http_client_contract PRIVATE /W4 /utf-8)
endif()
add_test(NAME http_client_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/../../integration_tests/test_http_client.py
    $<TARGET_FILE:chwell_http_client_contract>)
set_tests_properties(http_client_contract PROPERTIES TIMEOUT 45)
